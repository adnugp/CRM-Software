// src/lib/trySave.ts
import { toast } from '@/hooks/use-toast';

/**
 * Runs a save. On failure shows "Not saved" and returns false, so the form can
 * stay open and nothing typed is lost (e.g. a read-only company).
 */
export async function trySave(save: () => unknown): Promise<boolean> {
  try {
    await save();
    return true;
  } catch (error) {
    toast({
      title: 'Not saved',
      description: error instanceof Error ? error.message : 'Something went wrong. Please try again.',
      variant: 'destructive',
    });
    return false;
  }
}
```

// scripts/lib/migration.mjs
// Pure helpers for moving the old single-company layout (top-level collections)
// into one workspace. They work on Firestore REST documents ({ name, fields }),
// so dates, numbers and maps pass through untouched.

export const BUSINESS_COLLECTIONS = [
  'projects', 'tenders', 'employees', 'registrations', 'payments', 'subscriptions', 'partners', 'files',
];
export const COLLECTIONS = [...BUSINESS_COLLECTIONS, 'users'];
export const WORKSPACE_ID = 'grow-plus';
export const WORKSPACE_NAME = 'Grow Plus Technologies';
export const OWNER_EMAIL = 'mohamedismayil@gptechnologies.ae';
export const STAFF_ROLES = ['admin', 'manager', 'user'];
export const ROLES = [...STAFF_ROLES, 'client'];

export const docId = (doc) => doc.name.split('/').pop();
export const str = (doc, field) => doc.fields?.[field]?.stringValue;
const sv = (v) => ({ stringValue: v });

export function checkBackup(backup) {
  const missing = COLLECTIONS.filter((c) => !Array.isArray(backup?.collections?.[c]));
  if (missing.length) throw new Error(`Backup is missing: ${missing.join(', ')}`);
}

/** The record's fields without the old company tag. */
export function convertRecord(collection, doc) {
  const fields = { ...(doc.fields ?? {}) };
  delete fields.belongsTo;
  if (collection === 'files') delete fields.company;
  return fields;
}

export const countStaff = (users) => users.filter((u) => STAFF_ROLES.includes(str(u, 'role'))).length;

export const findOwner = (users, email = OWNER_EMAIL) =>
  users.find((u) => (str(u, 'email') ?? '').toLowerCase() === email.toLowerCase());

export function workspaceFields(users, now) {
  return {
    name: sv(WORKSPACE_NAME),
    divisions: { arrayValue: {} },
    status: sv('active'),
    plan: sv('custom'),
    staffLimit: { integerValue: '0' },
    staffCount: { integerValue: String(countStaff(users)) },
    createdAt: { timestampValue: now.toISOString() },
  };
}

/** Every write for documents:commit, in order. Safe to run twice. */
export function buildWrites(backup, { project, now }) {
  checkBackup(backup);
  const base = `projects/${project}/databases/(default)/documents`;
  const ws = `${base}/workspaces/${WORKSPACE_ID}`;
  const users = backup.collections.users;
  const wsFields = workspaceFields(users, now);

  // The mask keeps a logo or colour set after a first run.
  const writes = [{ update: { name: ws, fields: wsFields }, updateMask: { fieldPaths: Object.keys(wsFields) } }];
  for (const col of BUSINESS_COLLECTIONS) {
    for (const d of backup.collections[col]) {
      writes.push({ update: { name: `${ws}/${col}/${docId(d)}`, fields: convertRecord(col, d) } });
    }
  }
  for (const u of users) {
    writes.push({
      update: { name: `${base}/users/${docId(u)}`, fields: { workspaceId: sv(WORKSPACE_ID) } },
      updateMask: { fieldPaths: ['workspaceId'] },
    });
  }
  const owner = findOwner(users);
  if (owner) {
    writes.push({
      update: {
        name: `${base}/platformOwners/${docId(owner)}`,
        fields: { email: sv(OWNER_EMAIL), createdAt: { timestampValue: now.toISOString() } },
      },
    });
  }
  return writes;
}

/** What apply would do, plus anything that would break after the move. */
export function planReport(backup) {
  checkBackup(backup);
  const { users, projects } = backup.collections;
  const owner = findOwner(users);
  const warnings = [];

  for (const u of users) {
    const role = str(u, 'role');
    const email = str(u, 'email') ?? docId(u);
    if (!role) warnings.push(`${email} has no role (they will see no data).`);
    else if (!ROLES.includes(role)) warnings.push(`${email} has role "${role}", which is not one of ${ROLES.join(', ')}.`);
    if (role === 'client') {
      const org = str(u, 'organizationId');
      const linked = projects.some((p) => str(p, 'clientId') === docId(u) || (org && str(p, 'organizationId') === org));
      if (!linked) warnings.push(`Client ${email} has no linked projects (they will see none).`);
    }
  }
  for (const p of projects) {
    if (str(p, 'clientName') && !str(p, 'clientId') && !str(p, 'organizationId')) {
      warnings.push(`Project ${docId(p)} names a client but isn't linked to one (no clientId or organizationId).`);
    }
  }
  for (const col of BUSINESS_COLLECTIONS) {
    const n = backup.collections[col].filter((d) => !d.fields?.createdAt).length;
    if (n) warnings.push(`${col}: ${n} record${n === 1 ? '' : 's'} without createdAt (lists sort by it, so they won't show).`);
  }
  if (!owner) warnings.push(`Super admin ${OWNER_EMAIL} was not found among users.`);

  return {
    counts: Object.fromEntries(COLLECTIONS.map((c) => [c, backup.collections[c].length])),
    staffCount: countStaff(users),
    owner: owner ? str(owner, 'email') : null,
    users: users.map((u) => ({ email: str(u, 'email') ?? docId(u), role: str(u, 'role') ?? '', owner: u === owner })),
    warnings,
  };
}
```

// scripts/lib/firestore-rest.mjs
// Firestore over REST, the same way for the local emulator and the live project.
// Live uses no credentials: it only works while the old open rules are deployed.

// The live web API key is public (it's in every page of the app).
const LIVE_API_KEY = 'AIzaSyBcCR3upbuvG1fQFUOE9qLnITa8tk41cXI';

export function target(name) {
  if (name === 'emulator') {
    return { name, project: 'demo-crm', root: 'http://127.0.0.1:8085/v1', key: '',
      headers: { 'Content-Type': 'application/json', Authorization: 'Bearer owner' } };
  }
  if (name === 'live') {
    return { name, project: 'crm-gpt-2026', root: 'https://firestore.googleapis.com/v1', key: LIVE_API_KEY,
      headers: { 'Content-Type': 'application/json' } };
  }
  throw new Error(`Unknown target "${name}". Use emulator or live.`);
}

const documents = (t) => `${t.root}/projects/${t.project}/databases/(default)/documents`;
const withKey = (t, url) => (t.key ? `${url}${url.includes('?') ? '&' : '?'}key=${t.key}` : url);

async function send(url, init) {
  const res = await fetch(url, init);
  if (!res.ok) throw new Error(`${init?.method ?? 'GET'} ${url.replace(/key=[^&]+/, 'key=…')} -> ${res.status} ${await res.text()}`);
  return res.json();
}

/** Every document in a top-level collection. */
export async function listAll(t, collection) {
  const docs = [];
  let pageToken = '';
  do {
    const page = pageToken ? `&pageToken=${encodeURIComponent(pageToken)}` : '';
    const body = await send(withKey(t, `${documents(t)}/${collection}?pageSize=300${page}`), { headers: t.headers });
    docs.push(...(body.documents ?? []));
    pageToken = body.nextPageToken ?? '';
  } while (pageToken);
  return docs;
}

/** Apply writes in batches of 500 (Firestore's limit per commit). */
export async function commit(t, writes, onBatch = () => {}) {
  for (let i = 0; i < writes.length; i += 500) {
    const batch = writes.slice(i, i + 500);
    await send(withKey(t, `${documents(t)}:commit`), { method: 'POST', headers: t.headers, body: JSON.stringify({ writes: batch }) });
    onBatch(Math.min(i + 500, writes.length), writes.length);
  }
}

const flag = (name) => {
  const i = rest.indexOf(`--${name}`);
  return i === -1 ? undefined : rest[i + 1];
};
const file = rest.find((a) => !a.startsWith('--') && !['live', 'emulator'].includes(a));
const stamp = () => new Date().toISOString().slice(0, 16).replace('T', '-').replace(':', '');
const load = (path) => {
  if (!path) throw new Error('Give the backup file, e.g. backups/live-2026-11-09-1030.json');
  const backup = JSON.parse(readFileSync(path, 'utf8'));
  checkBackup(backup);
  return backup;
};

async function backup() {
  const t = target(flag('source') ?? '');
  if (t.name === 'emulator') await assertEmulatorsRunning();
  const collections = {};
  for (const c of COLLECTIONS) {
    collections[c] = await listAll(t, c);
    console.log(`  ${c.padEnd(14)} ${collections[c].length}`);
  }
  mkdirSync('backups', { recursive: true });
  const path = `backups/${t.name}-${stamp()}.json`;
  writeFileSync(path, JSON.stringify({ source: t.name, takenAt: new Date().toISOString(), collections }, null, 1));
  console.log(`\nSaved ${path}. Nothing was changed.`);
}

function plan() {
  const r = planReport(load(file));
  console.log('Records to copy into workspaces/grow-plus:');
  for (const [c, n] of Object.entries(r.counts)) console.log(`  ${c.padEnd(14)} ${n}`);
  console.log(`\nWorkspace: Grow Plus Technologies, active, plan custom, no staff limit, ${r.staffCount} staff.`);
  console.log('\nPeople:');
  for (const u of r.users) console.log(`  ${(u.role || '(none)').padEnd(8)} ${u.email}${u.owner ? '  ← super admin' : ''}`);
  console.log(r.warnings.length ? `\nWarnings (${r.warnings.length}):` : '\nNo warnings.');
  for (const w of r.warnings) console.log(`  ! ${w}`);
  console.log('\nThis was a dry run. Nothing was changed.');
}

async function apply() {
  const t = target(flag('target') ?? '');
  const data = load(file);
  if (t.name === 'emulator') await assertEmulatorsRunning();
  if (t.name === 'live') {
    const rl = createInterface({ input: process.stdin, output: process.stdout });
    const answer = await rl.question(`This writes to the LIVE project. Type ${t.project} to continue: `);
    rl.close();
    if (answer.trim() !== t.project) return console.log('Stopped. Nothing was changed.');
  }
  const writes = buildWrites(data, { project: t.project, now: new Date() });
  await commit(t, writes, (done, total) => console.log(`  wrote ${done} / ${total}`));
  console.log(`\nDone: ${writes.length} writes to ${t.name}.`);

  if (rest.includes('--emulator-logins')) {
    if (t.name !== 'emulator') throw new Error('--emulator-logins only works with --target emulator.');
    for (const u of data.collections.users) {
      const email = str(u, 'email');
      if (email) await upsertLoginWithUid(docId(u), email, TEST_PASSWORD, str(u, 'name') ?? email);
    }
    console.log(`Local logins created for ${data.collections.users.length} people (password: TEST_PASSWORD in scripts/emulator-rest.mjs).`);
  }
}

const steps = { backup, plan, apply };
if (!steps[step]) {
  console.log('Usage: npm run migrate -- backup --source live|emulator | plan <file> | apply <file> --target emulator|live [--emulator-logins]');
  process.exit(1);
}
Promise.resolve(steps[step]()).catch((err) => {
  console.error(`\nFailed: ${err.message}\nNothing after this point was written. Fix the cause and run the same command again.`);
  process.exit(1);
});
